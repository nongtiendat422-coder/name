local TS = game:GetService("TeleportService")
local HS = game:GetService("HttpService")
local PLRS = game:GetService("Players")
local SG = game:GetService("StarterGui")
local Lighting = game:GetService("Lighting")
local CollectionService = game:GetService("CollectionService")
local WS = game:GetService("Workspace")
local RS = game:GetService("ReplicatedStorage")
local PID = game.PlaceId
local LP = PLRS.LocalPlayer
local SEA_PIDS = {2753915549, 4442272183, 7449423635}
local SEA_NAMES = {"First Sea", "Second Sea", "Third Sea"}
local CURRENT_SEA = 1
for i=1,3 do if SEA_PIDS[i] == PID then CURRENT_SEA = i break end end
local function notify(title, txt, dur)
    pcall(function()
        SG:SetCore("SendNotification", {
            Title = title,
            Text = txt,
            Duration = dur or 5,
            Icon = "rbxassetid://76020773425692"
        })
    end)
end
local function getLowPopServers(maxPlaying)
    maxPlaying = maxPlaying or 5
    local servers, cursor = {}, ""
    repeat
        local url = string.format("https://games.roblox.com/v1/games/%d/servers/Public?sortOrder=Asc&limit=100%s", PID, cursor ~= "" and "&cursor="..cursor or "")
        local success, res = pcall(HS.JSONDecode, HS, game:HttpGet(url))
        if not success or not res or not res.data then break end
        for _, srv in res.data do
            if srv.playing <= maxPlaying and srv.playing < srv.maxPlayers then
                table.insert(servers, srv)
            end
        end
        cursor = res.nextPageCursor or ""
    until cursor == ""
    table.sort(servers, function(a,b) return a.playing < b.playing end)
    return servers
end
local function hopLowPop(maxPlaying)
    local servers = getLowPopServers(maxPlaying)
    if #servers == 0 then
        notify("Hop", "No low pop servers! Random hop...", 3)
        TS:Teleport(PID)
        return true
    end
    for _, srv in servers do
        if srv.id ~= game.JobId then
            notify("Hop", string.format("Hop %d/%d...", srv.playing, srv.maxPlayers), 2)
            TS:TeleportToPlaceInstance(PID, srv.id, LP)
            return true
        end
    end
    notify("Hop", "All low pop taken! Random...", 3)
    TS:Teleport(PID)
    return true
end
local function waitLoadAndCheck(checkFunc, checkName)
    task.wait(8)
    local success, has = pcall(checkFunc)
    if success and has then
        notify("FOUND ✅", checkName .. " TÌM THẤY! Dừng hop.", 10)
        return true
    end
    return false
end
local function hasFullMoon() return Lighting.ClockTime == 0 end
local function hasMirage() 
    local wo = WS:FindFirstChild("_WorldOrigin")
    return wo and wo:FindFirstChild("Locations") and wo.Locations:FindFirstChild("Mirage Island")
end
local function hasCakePrince() 
    return WS.Enemies:FindFirstChild("Cake Prince") or RS:FindFirstChild("Cake Prince")
end
local function hasDoughKing() 
    return WS.Enemies:FindFirstChild("Dough King") or RS:FindFirstChild("Dough King")
end
local function hasLegendaryBerry()
    for _, berry in CollectionService:GetTagged("Berry") do
        local rarity = berry:FindFirstChild("Rarity")
        if rarity and rarity:IsA("StringValue") and rarity.Value == "Legendary" then
            return true
        end
    end
    return false
end

local MAX_HOPS_SLIDER = 100
local MAX_PLAYERS = 5
local HOP_DELAY = 2
local hopping = {lowpop=false, fullmoon=false, mirage=false, cake=false, dough=false, berry=false}
LowSec:NewToggle("LowPopToggle", "Hop Low Pop (<= " .. MAX_PLAYERS .. ")", function(state)
    hopping.lowpop = state
    if state then
        notify("Low Pop", "Bắt đầu hop low pop...", 4)
        task.spawn(function()
            local count = 0
            while hopping.lowpop and count < MAX_HOPS_SLIDER do
                count += 1
                if hopLowPop(MAX_PLAYERS) then
                    task.wait(HOP_DELAY)
                else
                    break
                end
            end
            hopping.lowpop = false
            notify("Low Pop", "Dừng (max hops)", 4)
        end)
    end
end)
