local tabs = {}
local currentTab = nil

local function createTab(name)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.2, 0, 0, 40)
    btn.Position = UDim2.new((#tabs)*0.2, 0, 0, 10)
    btn.BackgroundColor3 = Color3.fromRGB(40, 40, 60)
    btn.Text = name
    btn.TextColor3 = Color3.new(1,1,1)
    btn.TextScaled = true
    btn.Parent = Frame
    
    local content = Instance.new("ScrollingFrame")
    content.Size = UDim2.new(0.95, 0, 1, -100)
    content.Position = UDim2.new(0.025, 0, 0, 60)
    content.BackgroundTransparency = 1
    content.Visible = false
    content.ScrollBarThickness = 8
    content.Parent = Frame
    
    btn.MouseButton1Click:Connect(function()
        for _, tab in pairs(tabs) do tab.content.Visible = false end
        content.Visible = true
        currentTab = content
    end)
    
    table.insert(tabs, {button = btn, content = content})
    return content
end

local function addToggle(content, name, varName)
    local btn = Instance.new("TextButton")
    btn.Size = UDim2.new(0.9, 0, 0, 35)
    btn.Position = UDim2.new(0.05, 0, 0, #content:GetChildren() * 40)
    btn.BackgroundColor3 = Color3.fromRGB(50, 50, 70)
    btn.Text = name .. ": OFF"
    btn.TextColor3 = Color3.new(1,1,1)
    btn.TextScaled = true
    btn.Parent = content
    
    _G[varName] = false
    btn.MouseButton1Click:Connect(function()
        _G[varName] = not _G[varName]
        btn.Text = name .. ": " .. (_G[varName] and "ON" or "OFF")
        btn.BackgroundColor3 = _G[varName] and Color3.fromRGB(0, 255, 0) or Color3.fromRGB(50, 50, 70)
    end)
end

local function getQuest(level)
    if level < 15 then return "BanditQuest1", "Bandit", "1"
    elseif level < 30 then return "BanditQuest2", "Bandit", "2"
    else return "GreenZone", "GreenBandit", "1" end
end

spawn(function()
    while true do
        wait(0.2)
        pcall(function()
            if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then return end
            local hrp = player.Character.HumanoidRootPart
            local level = ReplicatedStorage.Remotes.CommF_:InvokeServer("getMyLvl")
            
            if _G.AutoFarmLV then
                local quest, enemy, num = getQuest(level)
                ReplicatedStorage.Remotes.CommF_:InvokeServer("StartQuest", quest, enemy, num)
                for _, mob in pairs(workspace.Enemies:GetChildren()) do
                    if mob.Name:find(enemy) and mob.Humanoid.Health > 0 then
                        hrp.CFrame = mob.HumanoidRootPart.CFrame * CFrame.new(0, 10, 0)
                        VirtualUser:ClickButton1(Vector2.new())
                    end
                end
            end
            
            if _G.AutoMastery then
                ReplicatedStorage.Remotes.CommF_:InvokeServer("BuyFruit", "Random Fruit") 
                for _, tool in pairs(player.Backpack:GetChildren()) do
                    if tool:IsA("Tool") then
                        tool.Parent = player.Character
                        VirtualUser:ClickButton1(Vector2.new())
                    end
                end
            end
            
            if _G.AutoBoss then
                ReplicatedStorage.Remotes.CommF_:InvokeServer("requestEntrance", Vector3.new(923.212890625, 126.976997375, 52863.587890625))
                for _, boss in pairs(workspace.Enemies:GetChildren()) do
                    if boss.Name:find("Boss") then
                        hrp.CFrame = boss.HumanoidRootPart.CFrame * CFrame.new(0, 10, 0)
                        VirtualUser:ClickButton1(Vector2.new())
                    end
                end
            end
            
            if _G.FruitSniper then
                for _, fruit in pairs(workspace:GetChildren()) do
                    if fruit:IsA("Tool") and fruit:FindFirstChild("Handle") then
                        hrp.CFrame = fruit.Handle.CFrame
                        firetouchinterest(hrp, fruit.Handle, 0)
                        wait(0.1)
                        firetouchinterest(hrp, fruit.Handle, 1)
                    end
                end
            end
            
            if _G.AutoKitsune then
                hrp.CFrame = CFrame.new(-13200, 189, -7920)
                ReplicatedStorage.Remotes.CommF_:InvokeServer("KitsuneQuest")
            end
            
            if _G.ESPAll then
                for _, obj in pairs(workspace:GetChildren()) do
                    if obj.Name:find("Fruit") or obj.Name:find("Chest") or obj.Name:find("Boss") or obj.Name:find("Kitsune") then
                        local gui = obj:FindFirstChild("ESP")
                        if not gui then
                            gui = Instance.new("BillboardGui", obj)
                            gui.Name = "ESP"
                            gui.Size = UDim2.new(0, 100, 0, 50)
                            gui.AlwaysOnTop = true
                            local label = Instance.new("TextLabel", gui)
                            label.Text = obj.Name
                            label.TextColor3 = Color3.new(1,0,0)
                            label.BackgroundTransparency = 1
                            label.Size = UDim2.new(1,0,1,0)
                        end
                    end
                end
            end
        end)
    end
end)
