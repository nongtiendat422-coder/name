-- Load normal emotes from ReplicatedStorage.Assets.Emotes
local function loadNormalEmotes()
    local success, assets = pcall(function() return ReplicatedStorage:WaitForChild("Assets", 10) end)
    if not success then return end
    local success2, emotesFolder = pcall(function() return assets:WaitForChild("Emotes", 5) end)
    if not success2 then return end
    
    for _, module in pairs(emotesFolder:GetChildren()) do
        if module:IsA("ModuleScript") then
            local success3, data = pcall(require, module)
            if success3 and data.Animation then
                local animId = data.Animation.AnimationId:gsub("rbxassetid://", "")
                local soundId = data.Sound and data.Sound.SoundId:gsub("rbxassetid://", "") or nil
                normalEmotes[module.Name] = {anim = animId, sound = soundId}
            end
        end
    end
end

local function stopCurrentEmote()
    if currentTrack then
        currentTrack:Stop()
        currentTrack = nil
    end
    if currentSound then
        currentSound:Stop()
        currentSound = nil
    end
end

local function playEmote(emoteName, isVip)
    local emotesTable = isVip and vipEmotes or normalEmotes
    local data = emotesTable[emoteName]
    if not data or not data.anim then return end

    local character = player.Character
    if not character then return end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    if not humanoid then return end
    local rootPart = character:FindFirstChild("HumanoidRootPart")
    if not rootPart then return end

    stopCurrentEmote()

    -- Play animation
    local anim = Instance.new("Animation")
    anim.AnimationId = "rbxassetid://" .. data.anim
    currentTrack = humanoid:LoadAnimation(anim)
    currentTrack:Play()

    -- Play sound if available
    if data.sound then
        currentSound = Instance.new("Sound")
        currentSound.SoundId = "rbxassetid://" .. data.sound
        currentSound.Volume = 0.5
        currentSound.Parent = rootPart
        currentSound:Play()
        currentSound.Ended:Connect(function()
            currentSound:Destroy()
            currentSound = nil
        end)
    end
end

local function createButtons(scrollingFrame, emotesTable, isVip)
    local listLayout = scrollingFrame:FindFirstChild("UIListLayout") or Instance.new("UIListLayout")
    listLayout.SortOrder = Enum.SortOrder.LayoutOrder
    listLayout.Padding = UDim.new(0, 5)
    listLayout.Parent = scrollingFrame

    local i = 0
    for name, _ in pairs(emotesTable) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(1, -10, 0, 45)
        btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
        btn.Text = (isVip and "🔥 " or "🎭 ") .. name
        btn.TextColor3 = Color3.fromRGB(255, 255, 255)
        btn.TextScaled = true
        btn.Font = Enum.Font.GothamSemibold
        btn.LayoutOrder = i
        btn.Parent = scrollingFrame

        local btnCorner = Instance.new("UICorner")
        btnCorner.CornerRadius = UDim.new(0, 8)
        btnCorner.Parent = btn

        local btnStroke = Instance.new("UIStroke")
        btnStroke.Color = isVip and Color3.fromRGB(255, 100, 100) or Color3.fromRGB(100, 100, 100)
        btnStroke.Thickness = 1
        btnStroke.Parent = btn

        btn.MouseButton1Click:Connect(function()
            playEmote(name, isVip)
        end)

        btn.MouseEnter:Connect(function()
            TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(60, 60, 60)}):Play()
        end)
        btn.MouseLeave:Connect(function()
            TweenService:Create(btn, TweenInfo.new(0.2), {BackgroundColor3 = Color3.fromRGB(40, 40, 40)}):Play()
        end)

        i = i + 1
    end

    listLayout:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        scrollingFrame.CanvasSize = UDim2.new(0, 0, 0, listLayout.AbsoluteContentSize.Y + 10)
    end)
end
